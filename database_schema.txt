-- Base de Datos para Sistema de Gestión de Atención Hospitalaria
-- SQL Server Management Studio 2022

-- Crear la base de datos
CREATE DATABASE GestionAtencionHospital;
GO

USE GestionAtencionHospital;
GO

-- Tabla Usuario
-- Roles: 'Administrador', 'Recepcionista', 'Medico'
CREATE TABLE Usuario (
    UsuarioID INT IDENTITY(1,1) PRIMARY KEY,
    Nombre NVARCHAR(100) NOT NULL,
    Email NVARCHAR(100) UNIQUE NOT NULL,
    Password NVARCHAR(255) NOT NULL,
    Rol NVARCHAR(20) NOT NULL CHECK (Rol IN ('Administrador', 'Recepcionista', 'Medico'))
);

-- Tabla Paciente
CREATE TABLE Paciente (
    PacienteID INT IDENTITY(1,1) PRIMARY KEY,
    Nombre NVARCHAR(100) NOT NULL,
    Apellido NVARCHAR(100) NOT NULL,
    FechaNacimiento DATE NOT NULL,
    Direccion NVARCHAR(255),
    Telefono NVARCHAR(20),
    Email NVARCHAR(100),
    Expediente NVARCHAR(20) UNIQUE, -- EXP-0000-0000
    Cedula NVARCHAR(20),
    Sexo NVARCHAR(10),
    MotivoConsulta NVARCHAR(500),
    Sintomas NVARCHAR(500),
    TiempoSintomas NVARCHAR(100),
    Alergias NVARCHAR(500),
    MedicamentosActuales NVARCHAR(500),
    Urgencia INT DEFAULT 3, -- 1: Alta, 2: Media, 3: Baja
    Atendido BIT DEFAULT 0,
    FechaRegistro DATETIME DEFAULT GETDATE(),
    FechaAtencion DATETIME NULL,
    UsuarioRegistroID INT,
    FOREIGN KEY (UsuarioRegistroID) REFERENCES Usuario(UsuarioID)
);

-- Tabla Medico (especialización de Usuario)
CREATE TABLE Medico (
    MedicoID INT PRIMARY KEY,
    Especialidad NVARCHAR(100) NOT NULL,
    FOREIGN KEY (MedicoID) REFERENCES Usuario(UsuarioID)
);

-- Tabla Consulta
CREATE TABLE Consulta (
    ConsultaID INT IDENTITY(1,1) PRIMARY KEY,
    PacienteID INT NOT NULL,
    MedicoID INT NOT NULL,
    Fecha DATE NOT NULL,
    Hora TIME NOT NULL,
    Tipo NVARCHAR(20) NOT NULL CHECK (Tipo IN ('Cita', 'Urgencia')),
    Prioridad INT CHECK (Prioridad BETWEEN 1 AND 5), -- 1: Baja, 5: Alta
    FOREIGN KEY (PacienteID) REFERENCES Paciente(PacienteID),
    FOREIGN KEY (MedicoID) REFERENCES Medico(MedicoID)
);

-- Tabla HistorialClinico
CREATE TABLE HistorialClinico (
    HistorialID INT IDENTITY(1,1) PRIMARY KEY,
    ConsultaID INT NOT NULL,
    Diagnostico NVARCHAR(500) NOT NULL,
    Causa NVARCHAR(500),
    Tratamiento NVARCHAR(500),
    Fecha DATETIME DEFAULT GETDATE(),
    FOREIGN KEY (ConsultaID) REFERENCES Consulta(ConsultaID)
);

-- Normalización aplicada:
-- 1FN: Todos los atributos son atómicos.
-- 2FN: No hay dependencias parciales; claves primarias son simples o compuestas correctamente.
-- 3FN: No hay dependencias transitivas; atributos dependen solo de la clave primaria.

-- Procedimiento Almacenado 1: Registrar Paciente Nuevo
CREATE PROCEDURE RegistrarPaciente
    @Nombre NVARCHAR(100),
    @Apellido NVARCHAR(100),
    @FechaNacimiento DATE,
    @Direccion NVARCHAR(255),
    @Telefono NVARCHAR(20),
    @Email NVARCHAR(100),
    @UsuarioRegistroID INT
AS
BEGIN
    INSERT INTO Paciente (Nombre, Apellido, FechaNacimiento, Direccion, Telefono, Email, UsuarioRegistroID)
    VALUES (@Nombre, @Apellido, @FechaNacimiento, @Direccion, @Telefono, @Email, @UsuarioRegistroID);
END;
GO

-- Procedimiento Almacenado 2: Asignar Consulta
CREATE PROCEDURE AsignarConsulta
    @PacienteID INT,
    @MedicoID INT,
    @Fecha DATE,
    @Hora TIME,
    @Tipo NVARCHAR(20),
    @Prioridad INT
AS
BEGIN
    INSERT INTO Consulta (PacienteID, MedicoID, Fecha, Hora, Tipo, Prioridad)
    VALUES (@PacienteID, @MedicoID, @Fecha, @Hora, @Tipo, @Prioridad);
END;
GO

-- 1. JOIN: Pacientes con sus consultas y médicos asignados
SELECT P.Nombre + ' ' + P.Apellido AS Paciente, U.Nombre AS Medico, C.Fecha, C.Tipo
FROM Paciente P
JOIN Consulta C ON P.PacienteID = C.PacienteID
JOIN Medico M ON C.MedicoID = M.MedicoID
JOIN Usuario U ON M.MedicoID = U.UsuarioID;

-- 2. WHERE: Consultas de urgencia con prioridad alta
SELECT * FROM Consulta WHERE Tipo = 'Urgencia' AND Prioridad >= 4;

-- 3. GROUP BY: Número de consultas por médico
SELECT U.Nombre, COUNT(C.ConsultaID) AS NumeroConsultas
FROM Usuario U
JOIN Medico M ON U.UsuarioID = M.MedicoID
JOIN Consulta C ON M.MedicoID = C.MedicoID
GROUP BY U.Nombre;

-- 4. JOIN y WHERE: Historial clínico de pacientes con diagnóstico específico
SELECT P.Nombre + ' ' + P.Apellido AS Paciente, H.Diagnostico, H.Tratamiento
FROM Paciente P
JOIN Consulta C ON P.PacienteID = C.PacienteID
JOIN HistorialClinico H ON C.ConsultaID = H.ConsultaID
WHERE H.Diagnostico LIKE '%Hipertensión%';

-- 5. GROUP BY y agregación: Promedio de prioridad por tipo de consulta
SELECT Tipo, AVG(Prioridad) AS PromedioPrioridad
FROM Consulta
GROUP BY Tipo;
